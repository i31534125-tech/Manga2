import 'dart:io';
import 'package:path_provider/path_provider.dart';
import 'package:archive/archive.dart';

class ExportService {
  Future<String> exportChapter(List<String> imagePaths) async {
    final base=await getApplicationDocumentsDirectory();
    final outDir=Directory('${base.path}/MangaTranslate/exports');
    if(!outDir.existsSync()) outDir.createSync(recursive:true);

    final archive=Archive();
    for(final path in imagePaths){
      final file=File(path);
      if(!file.existsSync()) continue;
      final bytes=await file.readAsBytes();
      archive.addFile(ArchiveFile(
        path.split('/').last,
        bytes.length,
        bytes,
      ));
    }

    final zip=ZipEncoder().encode(archive);
    final out='${outDir.path}/capitulo_traduzido.zip';
    await File(out).writeAsBytes(zip as List<int>);
    return out;
  }
}

import 'dart:io';
import 'package:flutter/material.dart';
import 'package:image_picker/image_picker.dart';
import 'package:path_provider/path_provider.dart';
import '../models/page_job.dart';
import '../services/ocr_service.dart';
import '../services/translation_service.dart';
import '../services/compositor_service.dart';
import '../services/export_service.dart';
import 'review_screen.dart';
import 'reader_screen.dart';

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});
  @override State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  final picker = ImagePicker();
  final ocr = OcrService();
  final translation = TranslationService();
  final compositor = CompositorService();
  final exporter = ExportService();
  final pages = <PageJob>[];
  OcrScript script = OcrScript.japanese;
  bool busy = false;
  int tab = 0;

  String scriptName(OcrScript s) => switch (s) {
    OcrScript.japanese => 'Japonês',
    OcrScript.latin => 'Inglês',
    OcrScript.chinese => 'Chinês',
    OcrScript.devanagari => 'Devanágari',
    OcrScript.korean => 'Coreano',
  };

  String langCode(OcrScript s) => switch (s) {
    OcrScript.japanese => 'ja',
    OcrScript.latin => 'en',
    OcrScript.chinese => 'zh-CN',
    OcrScript.devanagari => 'hi',
    OcrScript.korean => 'ko',
  };

  Future<Directory> _originalDir() async {
    final root = await getApplicationDocumentsDirectory();
    final dir = Directory('${root.path}/MangaTranslate/original');
    await dir.create(recursive: true);
    return dir;
  }

  Future<Directory> _translatedDir() async {
    final root = await getApplicationDocumentsDirectory();
    final dir = Directory('${root.path}/MangaTranslate/translated');
    await dir.create(recursive: true);
    return dir;
  }

  Future<void> importImages() async {
    final picks = await picker.pickMultiImage(imageQuality: 100);
    if (picks.isEmpty) return;
    final dir = await _originalDir();
    for (var i = 0; i < picks.length; i++) {
      final src = File(picks[i].path);
      final name = '${DateTime.now().microsecondsSinceEpoch}_$i.jpg';
      final dst = File('${dir.path}/$name');
      await src.copy(dst.path);
      pages.add(PageJob(id: name, imagePath: dst.path));
    }
    setState(() {});
  }

  Future<void> processAll() async {
    if (pages.isEmpty || busy) return;
    setState(() => busy = true);
    for (final p in pages) {
      if (!mounted) break;
      await process(p);
    }
    if (mounted) setState(() => busy = false);
  }

  Future<void> process(PageJob p) async {
    setState(() => p.state = PageProcessState.ocr);
    try {
      final regions = await ocr.detectRegions(p.imagePath, script);
      for (final r in regions) {
        r.translatedText = await translation.translate(r.originalText, source: langCode(script));
      }
      p.regions = regions;
      p.state = PageProcessState.review;
    } catch (e) {
      p.state = PageProcessState.error;
      p.errorMessage = e.toString();
    }
    if (mounted) setState(() {});
  }

  Future<void> compose(PageJob p) async {
    if (p.regions.isEmpty) return;
    final dir = await _translatedDir();
    final baseName = p.id.replaceFirst(RegExp(r'\.(jpg|jpeg|png)$', caseSensitive: false), '');
    final out = '${dir.path}/$baseName.png';
    await compositor.compose(imagePath: p.imagePath, regions: p.regions, outputPath: out);
    p.outputPath = out;
    p.state = PageProcessState.composed;
    if (mounted) setState(() {});
  }

  String state(PageProcessState s) => switch (s) {
    PageProcessState.pending => 'Pronta para traduzir',
    PageProcessState.ocr => 'Lendo texto…',
    PageProcessState.translated => 'Revisada',
    PageProcessState.review => 'Aguardando revisão',
    PageProcessState.composed => 'Traduzida',
    PageProcessState.error => 'Erro',
  };

  Future<void> exportAll() async {
    final paths = pages.where((p) => p.outputPath != null).map((p) => p.outputPath!).toList();
    if (paths.isEmpty) return;
    final result = await exporter.exportChapter(paths);
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Capítulo exportado: $result')));
  }

  void openReader() {
    final translated = pages.where((p) => p.outputPath != null).toList();
    if (translated.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Traduza pelo menos uma página primeiro.')));
      return;
    }
    Navigator.push(context, MaterialPageRoute(builder: (_) => ReaderScreen(pages: translated)));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('MangaTranslate', style: TextStyle(fontWeight: FontWeight.bold)),
        actions: [
          IconButton(onPressed: exportAll, tooltip: 'Exportar', icon: const Icon(Icons.archive_outlined)),
          IconButton(onPressed: openReader, tooltip: 'Ler', icon: const Icon(Icons.menu_book_outlined)),
        ],
      ),
      body: IndexedStack(index: tab, children: [_library(), _settings()]),
      bottomNavigationBar: NavigationBar(
        selectedIndex: tab,
        onDestinationSelected: (v) => setState(() => tab = v),
        destinations: const [
          NavigationDestination(icon: Icon(Icons.library_books_outlined), selectedIcon: Icon(Icons.library_books), label: 'Biblioteca'),
          NavigationDestination(icon: Icon(Icons.settings_outlined), selectedIcon: Icon(Icons.settings), label: 'Configurações'),
        ],
      ),
      floatingActionButton: tab == 0 ? FloatingActionButton.extended(onPressed: busy ? null : importImages, icon: const Icon(Icons.add), label: const Text('Importar')) : null,
    );
  }

  Widget _library() {
    return Padding(
      padding: const EdgeInsets.fromLTRB(16, 12, 16, 90),
      child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Text('Minha biblioteca', style: Theme.of(context).textTheme.headlineSmall?.copyWith(fontWeight: FontWeight.bold)),
        const SizedBox(height: 4),
        Text(pages.isEmpty ? 'Importe as páginas do seu mangá para começar.' : '${pages.length} página(s) neste projeto.'),
        const SizedBox(height: 18),
        if (pages.isNotEmpty) ...[
          Card(child: Padding(padding: const EdgeInsets.all(16), child: Column(children: [
            Row(children: [
              const Icon(Icons.translate, size: 30),
              const SizedBox(width: 12),
              Expanded(child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
                const Text('Novo projeto', style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold)),
                Text('Idioma de origem: ${scriptName(script)}'),
              ])),
              IconButton(onPressed: openReader, icon: const Icon(Icons.menu_book)),
            ]),
            const SizedBox(height: 12),
            FilledButton.icon(onPressed: busy ? null : processAll, icon: const Icon(Icons.auto_awesome), label: Text(busy ? 'Processando…' : 'Traduzir todas as páginas'), style: FilledButton.styleFrom(minimumSize: const Size.fromHeight(48))),
          ]))),
          const SizedBox(height: 12),
          Expanded(child: ListView.separated(itemCount: pages.length, separatorBuilder: (_, __) => const SizedBox(height: 8), itemBuilder: (_, i) => _pageCard(pages[i], i))),
        ] else Expanded(child: Center(child: Column(mainAxisSize: MainAxisSize.min, children: [
          Icon(Icons.menu_book_outlined, size: 72, color: Theme.of(context).colorScheme.primary),
          const SizedBox(height: 12),
          const Text('Sua biblioteca está vazia', style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
          const SizedBox(height: 6),
          const Text('Toque em “Importar” para adicionar páginas.'),
        ]))),
      ]),
    );
  }

  Widget _pageCard(PageJob p, int i) {
    final path = p.outputPath ?? p.imagePath;
    return Card(child: ListTile(
      leading: ClipRRect(borderRadius: BorderRadius.circular(6), child: Image.file(File(path), width: 48, height: 64, fit: BoxFit.cover)),
      title: Text('Página ${i + 1}'),
      subtitle: Text(p.state == PageProcessState.error ? '${p.errorMessage}' : state(p.state), maxLines: 6),
      trailing: Wrap(spacing: 0, children: [
        if (p.state == PageProcessState.review) IconButton(icon: const Icon(Icons.rate_review_outlined), onPressed: () async { await Navigator.push(context, MaterialPageRoute(builder: (_) => ReviewScreen(page: p))); setState(() {}); }),
        if (p.state == PageProcessState.translated || p.state == PageProcessState.review) IconButton(icon: const Icon(Icons.auto_fix_high), onPressed: () => compose(p)),
        if (p.state == PageProcessState.composed) IconButton(icon: const Icon(Icons.menu_book_outlined), onPressed: openReader),
      ]),
    ));
  }

  Widget _settings() => ListView(padding: const EdgeInsets.all(16), children: [
    Text('Configurações', style: Theme.of(context).textTheme.headlineSmall?.copyWith(fontWeight: FontWeight.bold)),
    const SizedBox(height: 18),
    DropdownButtonFormField<OcrScript>(value: script, decoration: const InputDecoration(labelText: 'Idioma/escrita das páginas', border: OutlineInputBorder()), items: OcrScript.values.map((s) => DropdownMenuItem(value: s, child: Text(scriptName(s)))).toList(), onChanged: (v) => setState(() => script = v ?? OcrScript.japanese)),
    const SizedBox(height: 16),
    const Card(child: ListTile(leading: Icon(Icons.storage_outlined), title: Text('Armazenamento'), subtitle: Text('Os originais e traduções ficam na pasta privada do aplicativo.'))),
    const Card(child: ListTile(leading: Icon(Icons.shield_outlined), title: Text('Originais preservados'), subtitle: Text('A composição cria uma cópia traduzida sem alterar a página original.'))),
  ]);
}
