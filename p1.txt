import 'dart:io';
import 'dart:math';
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import 'package:image/image.dart' as img;
import '../models/text_region.dart';
import 'inpainting_service.dart';

class CompositorService {
  final InpaintingService inpainting=InpaintingService();

  Future<String> compose({
    required String imagePath,
    required List<TextRegion> regions,
    required String outputPath,
  }) async {
    final bytes=await File(imagePath).readAsBytes();
    final decoded=img.decodeImage(bytes);
    if(decoded==null) throw Exception('Imagem inválida.');

    final cleanRegions=regions.where((r)=>r.enabled).map((r)=>RectData(
      r.rect.left,r.rect.top,r.rect.right,r.rect.bottom)).toList();

    final cleaned=inpainting.removeText(decoded,cleanRegions);
    final cleanBytes=img.encodePng(cleaned);

    final recorder=ui.PictureRecorder();
    final canvas=Canvas(recorder,Rect.fromLTWH(0,0,cleaned.width.toDouble(),cleaned.height.toDouble()));
    final codec=await ui.instantiateImageCodec(cleanBytes);
    final frame=await codec.getNextFrame();
    canvas.drawImage(frame.image,Offset.zero,Paint());

    for(final r in regions.where((r)=>r.enabled)){
      final rect=r.rect;
      final paint=Paint()..color=Colors.white.withOpacity(
        r.type==RegionType.freeText ? 0.78 : 0.94,
      );
      canvas.drawRRect(
        RRect.fromRectAndRadius(
          rect,
          Radius.circular(r.type==RegionType.speechBubble?24:7),
        ),
        paint,
      );

      final tp=TextPainter(
        text:TextSpan(
          text:r.translatedText,
          style:TextStyle(
            color:Colors.black,
            fontSize:_font(r.translatedText,rect),
            fontWeight:FontWeight.w600,
          ),
        ),
        textAlign:TextAlign.center,
        textDirection:TextDirection.ltr,
      )..layout(maxWidth:rect.width-16);

      tp.paint(canvas,Offset(
        rect.center.dx-tp.width/2,
        rect.center.dy-tp.height/2,
      ));
    }

    final picture=recorder.endRecording();
    final result=await picture.toImage(cleaned.width,cleaned.height);
    final data=await result.toByteData(format:ui.ImageByteFormat.png);
    await File(outputPath).writeAsBytes(data!.buffer.asUint8List());
    return outputPath;
  }

  double _font(String text,Rect r){
    final base=(r.width*r.height)/(text.length.clamp(1,100)*20);
    var size=sqrt(base);
    if(size<11) size=11;
    if(size>42) size=42;
    return size;
  }
}

import 'dart:convert';
import 'package:http/http.dart' as http;

/// Tradução gratuita via MyMemory (sem chave de API, com limite diário).
/// Para usar outro serviço, troque apenas o método _one.
class TranslationService {
  static const _maxChunk = 450;

  Future<String> translate(String text, {String source = 'ja'}) async {
    final clean = text.replaceAll('\n', ' ').trim();
    if (clean.isEmpty) return '';
    final out = <String>[];
    for (var i = 0; i < clean.length; i += _maxChunk) {
      final end = (i + _maxChunk > clean.length) ? clean.length : i + _maxChunk;
      out.add(await _one(clean.substring(i, end), source));
    }
    return out.join(' ');
  }

  Future<String> _one(String text, String source) async {
    try {
      final uri = Uri.https('api.mymemory.translated.net', '/get', {
        'q': text,
        'langpair': '$source|pt-BR',
      });
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        final data = jsonDecode(utf8.decode(r.bodyBytes));
        if (data['responseStatus'] == 200) {
          final t = data['responseData']?['translatedText']?.toString();
          if (t != null && t.isNotEmpty) return t;
        }
      }
    } catch (_) {}
    return text;
  }
}

import 'dart:math';
import 'package:image/image.dart' as img;

class InpaintingService {
  /// Reconstrução simples para áreas de texto.
  /// Usa amostras da borda da região e uma suavização local.
  /// Não substitui inpainting neural.
  img.Image removeText(img.Image source, List<RectData> regions) {
    final out=img.Image.from(source);

    for(final r in regions){
      final left=max(0,r.left.round());
      final top=max(0,r.top.round());
      final right=min(out.width-1,r.right.round());
      final bottom=min(out.height-1,r.bottom.round());

      final samples=<img.Color>[];
      for(int x=left;x<=right;x+=max(1,(right-left)~/12+1)){
        if(top>0) samples.add(out.getPixel(x,top-1));
        if(bottom<out.height-1) samples.add(out.getPixel(x,bottom+1));
      }
      for(int y=top;y<=bottom;y+=max(1,(bottom-top)~/12+1)){
        if(left>0) samples.add(out.getPixel(left-1,y));
        if(right<out.width-1) samples.add(out.getPixel(right+1,y));
      }

      if(samples.isEmpty) continue;
      final avg=img.ColorRgb8(
        samples.map((c)=>c.r.toInt()).reduce((a,b)=>a+b)~/samples.length,
        samples.map((c)=>c.g.toInt()).reduce((a,b)=>a+b)~/samples.length,
        samples.map((c)=>c.b.toInt()).reduce((a,b)=>a+b)~/samples.length,
      );

      for(int y=top;y<=bottom;y++){
        for(int x=left;x<=right;x++){
          final edge=((x-left).abs()+(right-x).abs()+(y-top).abs()+(bottom-y).abs());
          final maxEdge=(right-left+bottom-top).toDouble();
          final blend=(edge/maxEdge).clamp(0.0,1.0);
          final old=out.getPixel(x,y);
          final rr=(old.r*(blend*0.15)+avg.r*(1-blend*0.15)).round();
          final gg=(old.g*(blend*0.15)+avg.g*(1-blend*0.15)).round();
          final bb=(old.b*(blend*0.15)+avg.b*(1-blend*0.15)).round();
          out.setPixelRgb(x,y,rr,gg,bb);
        }
      }
    }
    return out;
  }
}

class RectData {
  final double left,top,right,bottom;
  RectData(this.left,this.top,this.right,this.bottom);
}

import 'dart:ui';
import 'package:google_mlkit_text_recognition/google_mlkit_text_recognition.dart';
import '../models/text_region.dart';

enum OcrScript { japanese, latin, chinese, devanagari, korean }

class OcrService {
  TextRecognizer _recognizer(OcrScript s) {
    switch (s) {
      case OcrScript.japanese:
        return TextRecognizer(script: TextRecognitionScript.japanese);
      case OcrScript.latin:
        return TextRecognizer(script: TextRecognitionScript.latin);
      case OcrScript.chinese:
        return TextRecognizer(script: TextRecognitionScript.chinese);
      case OcrScript.devanagari:
        return TextRecognizer(script: TextRecognitionScript.devanagiri);
      case OcrScript.korean:
        return TextRecognizer(script: TextRecognitionScript.korean);
    }
  }

  Future<List<TextRegion>> detectRegions(String path, OcrScript script) async {
    final recognizer = _recognizer(script);
    try {
      final result = await recognizer.processImage(InputImage.fromFilePath(path));
      final regions=<TextRegion>[];

      for (final block in result.blocks) {
        for (final line in block.lines) {
          if (line.text.trim().isEmpty) continue;
          final r=line.boundingBox.inflate(8);
          final vertical=r.height > r.width*1.25;
          final type=vertical
              ? RegionType.speechBubble
              : (r.width > r.height*3 ? RegionType.freeText : RegionType.speechBubble);

          regions.add(TextRegion(
            originalText: line.text.trim(),
            translatedText: line.text.trim(),
            rect:r,
            type:type,
            vertical:vertical,
            confidence:0.75,
          ));
        }
      }

      // Agrupa linhas muito próximas.
      regions.sort((a,b)=>a.rect.top.compareTo(b.rect.top));
      final grouped=<TextRegion>[];
      for(final r in regions){
        if(grouped.isEmpty){ grouped.add(r); continue; }
        final last=grouped.last;
        final gap=r.rect.top-last.rect.bottom;
        final overlap=Rect.fromLTRB(
          r.rect.left>last.rect.left?r.rect.left:last.rect.left,
          0,
          r.rect.right<last.rect.right?r.rect.right:last.rect.right,
          1,
        ).width;
        if(gap<28 && overlap>0){
          final u=last.rect.expandToInclude(r.rect);
          last.rect=u;
          last.originalText='${last.originalText}\n${r.originalText}';
          last.translatedText=last.originalText;
        }else{
          grouped.add(r);
        }
      }
      return grouped;
    } finally {
      await recognizer.close();
    }
  }
}
